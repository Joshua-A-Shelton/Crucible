# The C++ core library, in two forms built from the same objects (ADR 0011):
#   crucible_core         shared, for JIT and editor builds; C# binds to it by name
#   crucible_core_static  static, for AOT builds (linked into one executable) and C++ tests

add_library(crucible_core_objects OBJECT)
crucible_set_compile_options(crucible_core_objects)

# crucible/Version.h is generated from the project version.
set(CRUCIBLE_GENERATED_INCLUDE_DIR "${CMAKE_CURRENT_BINARY_DIR}/include")
configure_file("${PROJECT_SOURCE_DIR}/cmake/Version.h.in" "${CRUCIBLE_GENERATED_INCLUDE_DIR}/crucible/Version.h" @ONLY)

target_include_directories(crucible_core_objects PUBLIC
    "${CMAKE_CURRENT_SOURCE_DIR}/include"
    "${CRUCIBLE_GENERATED_INCLUDE_DIR}")

# Every public header is given to the binding generator (ADR 0017); only
# CRUCIBLE_EXPOSE declarations produce bindings, so exposing something new
# takes one annotation and no build-file edit.
file(GLOB CRUCIBLE_CORE_HEADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/include/crucible/*.h")
crucible_generate_bindings(
    TARGET crucible_bindings
    OUTPUT_CPP "${CRUCIBLE_GENERATED_DIR}/native/Bindings.g.cpp"
    OUTPUT_CS "${CRUCIBLE_GENERATED_DIR}/managed/Bindings.g.cs"
    # Slag's headers too: RenderSchedule.h uses its render graph's types.
    INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/include" "${slag_SOURCE_DIR}/include"
    HEADERS ${CRUCIBLE_CORE_HEADERS})

target_sources(crucible_core_objects PRIVATE
    BinaryFile.cpp
    BindingProbe.cpp
    Checksum.cpp
    Core.cpp
    Demo.cpp
    Ecs.cpp
    Jobs.cpp
    Launch.cpp
    Math.cpp
    MathProbe.cpp
    Memory.cpp
    Module.cpp
    Platform.cpp
    Shader.cpp
    Systems.cpp
    Transform.cpp)

# Only CRUCIBLE_EXPORT symbols are visible from the shared library.
target_compile_definitions(crucible_core_objects PRIVATE CRUCIBLE_CORE_EXPORTS)
set_target_properties(crucible_core_objects PROPERTIES
    CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN ON)

target_link_libraries(crucible_core_objects PUBLIC slag::slag)

# Public, so tests and the launcher see the same SIMD path as the core.
if(CRUCIBLE_FORCE_SCALAR_SIMD)
    target_compile_definitions(crucible_core_objects PUBLIC CRUCIBLE_SIMD_SCALAR=1)
endif()
# SDL stays private: it never appears in a public header (ADR 0018).
target_link_libraries(crucible_core_objects PRIVATE SDL3::SDL3-static)
# Compression for binary files (ADR 0039).
target_link_libraries(crucible_core_objects PRIVATE libzstd_static)
# Loading native modules (ADR 0074).
target_link_libraries(crucible_core_objects PRIVATE ${CMAKE_DL_LIBS})

# Window drags on Wayland (ADR 0066): a connection of the platform's own,
# with its protocols generated by wayland-scanner (MIT; SDL's Wayland build
# needs it too) from SDL's pinned copies of the core and xdg-shell XML and
# the vendored xdg-toplevel-drag. libwayland-client (MIT) is loaded at run
# time. Without the scanner or headers, Linux builds drag with SDL alone
# (X11) and windows don't drag between each other on Wayland.
if(UNIX AND NOT APPLE)
    find_program(CRUCIBLE_WAYLAND_SCANNER wayland-scanner)
    find_path(CRUCIBLE_WAYLAND_INCLUDE wayland-client-core.h)
    if(CRUCIBLE_WAYLAND_SCANNER AND CRUCIBLE_WAYLAND_INCLUDE)
        set(waylandGenerated "${CMAKE_CURRENT_BINARY_DIR}/wayland")
        set(waylandSources)
        foreach(protocol
                "wayland|${sdl3_SOURCE_DIR}/wayland-protocols/wayland.xml"
                "xdg-shell|${sdl3_SOURCE_DIR}/wayland-protocols/xdg-shell.xml"
                "xdg-toplevel-drag-v1|${PROJECT_SOURCE_DIR}/third_party/wayland/xdg-toplevel-drag-v1.xml")
            string(REPLACE "|" ";" protocol "${protocol}")
            list(GET protocol 0 name)
            list(GET protocol 1 xml)
            add_custom_command(
                OUTPUT "${waylandGenerated}/${name}-protocol.h" "${waylandGenerated}/${name}-protocol.c"
                COMMAND "${CMAKE_COMMAND}" -E make_directory "${waylandGenerated}"
                COMMAND "${CMAKE_COMMAND}" "-DSCANNER=${CRUCIBLE_WAYLAND_SCANNER}" "-DXML=${xml}"
                        "-DHEADER=${waylandGenerated}/${name}-protocol.h" "-DCODE=${waylandGenerated}/${name}-protocol.c"
                        -P "${PROJECT_SOURCE_DIR}/cmake/WaylandProtocol.cmake"
                DEPENDS "${xml}" "${PROJECT_SOURCE_DIR}/cmake/WaylandProtocol.cmake"
                VERBATIM)
            list(APPEND waylandSources "${waylandGenerated}/${name}-protocol.h" "${waylandGenerated}/${name}-protocol.c")
        endforeach()
        # The generated code is included by PlatformWayland.cpp, not compiled alone.
        set_source_files_properties(${waylandSources} PROPERTIES HEADER_FILE_ONLY ON)
        target_sources(crucible_core_objects PRIVATE PlatformWayland.cpp ${waylandSources})
        target_include_directories(crucible_core_objects PRIVATE "${waylandGenerated}" "${CRUCIBLE_WAYLAND_INCLUDE}")
        target_compile_definitions(crucible_core_objects PRIVATE CRUCIBLE_PLATFORM_WAYLAND=1)
        target_link_libraries(crucible_core_objects PRIVATE ${CMAKE_DL_LIBS})
    else()
        message(STATUS "Crucible: wayland-scanner or headers not found; no Wayland window drags")
    endif()
    # Window moves on X11 (ADR 0066): libX11 (X11 license, MIT-style) loaded at
    # run time; only its headers at build time, which SDL's X11 build needs too.
    find_path(CRUCIBLE_X11_INCLUDE X11/Xlib.h)
    if(CRUCIBLE_X11_INCLUDE)
        target_sources(crucible_core_objects PRIVATE PlatformX11.cpp)
        target_include_directories(crucible_core_objects PRIVATE "${CRUCIBLE_X11_INCLUDE}")
        target_compile_definitions(crucible_core_objects PRIVATE CRUCIBLE_PLATFORM_X11=1)
        target_link_libraries(crucible_core_objects PRIVATE ${CMAKE_DL_LIBS})
    endif()
endif()
if(WIN32)
    target_sources(crucible_core_objects PRIVATE PlatformWindows.cpp)
endif()

# The renderer (ADR 0015, 0048), with the engine's shaders embedded, and the
# generated bindings (which call into it): built after shaderfax, which links
# the objects above and compiles those shaders (ADR 0048's build order).
add_library(crucible_render_objects OBJECT
    Camera.cpp
    DrawList.cpp
    Font.cpp
    GlyphCache.cpp
    GlyphUpload.cpp
    Lights.cpp
    Materials.cpp
    Render.cpp
    RenderAssets.cpp
    RenderCommands.cpp
    RenderSchedule.cpp
    RenderSteps.cpp
    Shaping.cpp
    Shadows.cpp
    "${CRUCIBLE_GENERATED_DIR}/native/Bindings.g.cpp")
crucible_set_compile_options(crucible_render_objects)
target_compile_definitions(crucible_render_objects PRIVATE CRUCIBLE_CORE_EXPORTS)
set_target_properties(crucible_render_objects PROPERTIES
    CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN ON)
target_link_libraries(crucible_render_objects PUBLIC crucible_core_objects slag::rendergraph)
# Text (ADR 0057): never in a public header.
target_link_libraries(crucible_render_objects PRIVATE freetype crucible_harfbuzz crucible_msdfgen)
# The C++ mirrors of the shader library's [GpuShared] structs (ViewConstants).
crucible_add_shader(crucible_render_objects SOURCE "${PROJECT_SOURCE_DIR}/shaders/crucible/shared.slang"
    CPP_NAMESPACE crucible::render CS_NAMESPACE Crucible.Render)
crucible_embed_shaders(crucible_render_objects
    OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/EngineShaders.g.cpp"
    SOURCES "${PROJECT_SOURCE_DIR}/shaders/crucible/mesh.slang"
        "${PROJECT_SOURCE_DIR}/shaders/crucible/default.slang"
        "${PROJECT_SOURCE_DIR}/shaders/crucible/tonemap.slang"
        "${PROJECT_SOURCE_DIR}/shaders/crucible/clusters.slang"
        "${PROJECT_SOURCE_DIR}/shaders/crucible/sky.slang"
        "${PROJECT_SOURCE_DIR}/shaders/crucible/bloom.slang"
        "${PROJECT_SOURCE_DIR}/shaders/crucible/ui.slang"
        "${PROJECT_SOURCE_DIR}/shaders/crucible/editor.slang"
        "${PROJECT_SOURCE_DIR}/shaders/crucible/editorOverlay.slang"
    # The C# mirrors of their [GpuShared] structs (DefaultMaterial), for
    # Crucible.Engine.
    CS_OUTPUT_DIR "${CRUCIBLE_GENERATED_DIR}/managed/engine-shaders")

# For shaderfax only: no embedded shaders.
add_library(crucible_engine_shaders_stub OBJECT EngineShadersStub.cpp)
crucible_set_compile_options(crucible_engine_shaders_stub)
target_link_libraries(crucible_engine_shaders_stub PRIVATE crucible_core_objects)

add_library(crucible_core SHARED)
add_library(crucible::core ALIAS crucible_core)
target_link_libraries(crucible_core PUBLIC crucible_core_objects crucible_render_objects)
if(NOT WIN32 AND NOT APPLE)
    # Keep Slag's symbols (and any other static library's) out of the
    # shared library's export table.
    target_link_options(crucible_core PRIVATE "LINKER:--exclude-libs,ALL")
endif()

add_library(crucible_core_static STATIC)
add_library(crucible::core_static ALIAS crucible_core_static)
target_link_libraries(crucible_core_static PUBLIC crucible_core_objects crucible_render_objects)
